// p5.js: 키(UP, DOWN)로 정숫값 송신 

let port = new p5.WebSerial();
let freq = 0;

function setup() {
  createCanvas(500, 300);
  port.getPorts();
  port.on('noport', makePortButton);
  port.on('portavailable', openPort);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textAlign(CENTER, CENTER);
  textSize(50);
  text('주파수: ' + freq, width/2, height/2);
  let value = map(freq, 100, 500, height, 0);
  fill(255, 0, 0, 200);
  ellipse(width/2, value, 50, 50);
}

function keyPressed() {
  if (keyCode === UP_ARROW) {
    freq += 100;
  } else if (keyCode === DOWN_ARROW) {
    freq -= 100;
  }
  freq = constrain(freq, 100, 500);
  port.write(freq + '\n');
}

function makePortButton() {
  let portButton = createButton('포트 선택');
  portButton.mousePressed(choosePort);
  function choosePort() {
    port.requestPort();
  }
}

function openPort() {
  port.open().then(initiateSerial);
  function initiateSerial() {
    console.log('포트 오픈');
  }
}
