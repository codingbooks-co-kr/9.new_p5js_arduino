// 라디안(radian)을 이용하여 다양한 각의 원호를 그리기

function setup() {
  createCanvas(500, 200);
  background(200);

  arc(100, 100, 80, 80, 0, PI/2);	// 원호의 중심: (100, 100), 폭·높이: (80, 80), 각(rad) 범위: 0∼π/2
  arc(200, 100, 80, 80, 0, PI);    	// 원호의 각(rad) 범위: 0∼π
  arc(300, 100, 80, 80, 0, PI*3/2);	// 원호의 각(rad) 범위: 0∼3π/2
  arc(400, 100, 80, 80, 0, PI*2);  	// 원호의 각(rad) 범위: 0∼2π
}
