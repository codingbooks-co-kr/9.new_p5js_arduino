function setup() {
  console.log(int(true));			// 논리
  console.log(int(false));			// 논리
  console.log(int('10'));			// 문자열
  console.log(int(20.5));			// 실수
  console.log(int([true, false, '10', 20.5]));  	// 배열

  console.log(float(true));		// 논리
  console.log(float(false));		// 논리
  console.log(float('10.5'));		// 문자열
  console.log(float([true, false, '10.5', '20.5']));	// 배열

  console.log(str(10.5));			// 실수
  console.log(str(10)); 			// 정수
  console.log(str('10'));			// 문자열
  console.log(str(true));			// 논리
  console.log(str(false));		// 논리
  console.log(str([true, '10.5', 10.5]));	// 배열
}
